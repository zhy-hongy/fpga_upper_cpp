cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(FPGASDK VERSION 2.0.0 LANGUAGES CXX)
    include(CTest)
endif()
if(MSVC AND DEFINED ENV{FPGA_MSVC_INCLUDE_PREFIX})
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "$ENV{FPGA_MSVC_INCLUDE_PREFIX}")
endif()
option(FPGA_BUILD_DEMO "Build the Qt Widgets SDK demonstration" ON)
option(FPGA_BUILD_VISION "Build optional OpenCV laser/alignment/chessboard processing" ON)
set(FPGA_PYTHON_REFERENCE_ROOT "" CACHE PATH "Existing Python upper-computer source tree for differential tests")
set(FPGA_TEST_PYTHON "" CACHE FILEPATH "Python interpreter with the reference application's dependencies")
find_package(Threads REQUIRED)
add_library(fpga_sdk SHARED src/wire.cpp src/data.cpp src/device.cpp src/simulator.cpp src/upgrade.cpp src/analysis.cpp src/tools.cpp)
add_library(FPGA::SDK ALIAS fpga_sdk)
target_compile_features(fpga_sdk PUBLIC cxx_std_17)
target_include_directories(fpga_sdk PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(fpga_sdk PRIVATE Threads::Threads)
if(WIN32)
    target_link_libraries(fpga_sdk PRIVATE ws2_32)
    set_target_properties(fpga_sdk PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
endif()
if(FPGA_BUILD_VISION)
    find_package(OpenCV REQUIRED COMPONENTS core imgproc calib3d)
    add_library(fpga_vision SHARED src/vision.cpp)
    target_compile_features(fpga_vision PUBLIC cxx_std_17)
    target_include_directories(fpga_vision PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
    target_link_libraries(fpga_vision PUBLIC ${OpenCV_LIBS})
    target_include_directories(fpga_vision PUBLIC $<BUILD_INTERFACE:${OpenCV_INCLUDE_DIRS}>)
    set_target_properties(fpga_vision PROPERTIES OUTPUT_NAME FPGAVision EXPORT_NAME Vision WINDOWS_EXPORT_ALL_SYMBOLS ON)
    if(MSVC)
        target_compile_options(fpga_vision PRIVATE /utf-8 /W4)
    endif()
    if(BUILD_TESTING)
        add_executable(fpga_vision_test tests/vision_test.cpp)
        target_link_libraries(fpga_vision_test PRIVATE fpga_vision)
        if(MSVC)
            target_compile_options(fpga_vision_test PRIVATE /utf-8)
        endif()
        add_test(NAME sdk_vision COMMAND fpga_vision_test)
    endif()
    install(TARGETS fpga_vision EXPORT FPGAVisionTargets RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
    install(EXPORT FPGAVisionTargets NAMESPACE FPGA:: DESTINATION lib/cmake/FPGASDK)
endif()
set_target_properties(fpga_sdk PROPERTIES OUTPUT_NAME FPGASDK VERSION 2.0.0 SOVERSION 2 EXPORT_NAME SDK)
if(MSVC)
    target_compile_options(fpga_sdk PRIVATE /utf-8 /W4)
else()
    target_compile_options(fpga_sdk PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING AND FPGA_PYTHON_REFERENCE_ROOT AND FPGA_TEST_PYTHON)
    add_test(NAME sdk_python_wire_diff COMMAND "${FPGA_TEST_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/python_wire_diff.py" --python-root "${FPGA_PYTHON_REFERENCE_ROOT}" --wire-test $<TARGET_FILE:fpga_wire_test>)
    add_test(NAME sdk_python_capture_diff COMMAND "${FPGA_TEST_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/data_reference_compare.py" --python-root "${FPGA_PYTHON_REFERENCE_ROOT}" --probe $<TARGET_FILE:fpga_data_test>)
    add_test(NAME sdk_python_analysis_diff COMMAND "${FPGA_TEST_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/analysis_reference_compare.py" --python-root "${FPGA_PYTHON_REFERENCE_ROOT}" --probe $<TARGET_FILE:fpga_analysis_test>)
    set_tests_properties(sdk_python_wire_diff sdk_python_capture_diff sdk_python_analysis_diff PROPERTIES TIMEOUT 120)
    if(FPGA_BUILD_VISION)
        add_test(NAME sdk_python_vision_diff COMMAND "${FPGA_TEST_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/vision_reference_compare.py" --python-root "${FPGA_PYTHON_REFERENCE_ROOT}" --vision-test $<TARGET_FILE:fpga_vision_test>)
        set_tests_properties(sdk_python_vision_diff PROPERTIES TIMEOUT 120)
    endif()
endif()
if(BUILD_TESTING)
    foreach(name wire data device upgrade analysis tools stream)
        add_executable(fpga_${name}_test tests/${name}_test.cpp)
        target_link_libraries(fpga_${name}_test PRIVATE FPGA::SDK Threads::Threads)
        if(WIN32)
            target_link_libraries(fpga_${name}_test PRIVATE ws2_32)
        endif()
        if(MSVC)
            target_compile_options(fpga_${name}_test PRIVATE /utf-8)
        endif()
        add_test(NAME sdk_${name} COMMAND fpga_${name}_test)
        set_tests_properties(sdk_${name} PROPERTIES TIMEOUT 45)
    endforeach()
endif()
add_executable(fpga_hardware_probe tests/hardware_probe.cpp)
target_link_libraries(fpga_hardware_probe PRIVATE FPGA::SDK)
if(MSVC)
    target_compile_options(fpga_hardware_probe PRIVATE /utf-8)
endif()
add_executable(fpga_hardware_roundtrip tests/hardware_roundtrip.cpp)
target_link_libraries(fpga_hardware_roundtrip PRIVATE FPGA::SDK)
if(MSVC)
    target_compile_options(fpga_hardware_roundtrip PRIVATE /utf-8)
endif()
if(FPGA_BUILD_DEMO)
    find_package(Qt6 REQUIRED COMPONENTS Widgets)
    add_executable(fpga_sdk_demo ../demo_v2/main.cpp ../demo_v2/main_window.cpp ../demo_v2/main_window.hpp)
    set_target_properties(fpga_sdk_demo PROPERTIES AUTOMOC ON WIN32_EXECUTABLE TRUE)
    target_link_libraries(fpga_sdk_demo PRIVATE FPGA::SDK Qt6::Widgets Threads::Threads)
    if(MSVC)
        target_compile_options(fpga_sdk_demo PRIVATE /utf-8)
    endif()
    if(WIN32)
        # vcpkg supplies plugin DLLs separately from Qt's linked runtime DLLs.
        get_filename_component(_qt_prefix "${Qt6_DIR}/../.." ABSOLUTE)
        set(_qt_plugins "${_qt_prefix}/$<IF:$<CONFIG:Debug>,debug/,>Qt6/plugins")
        add_custom_command(TARGET fpga_sdk_demo POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_directory "${_qt_plugins}/platforms" "$<TARGET_FILE_DIR:fpga_sdk_demo>/platforms")
        get_target_property(_qt_qmake Qt6::qmake IMPORTED_LOCATION)
        get_filename_component(_qt_bin "${_qt_qmake}" DIRECTORY)
        find_program(_qt_deploy NAMES windeployqt PATHS "${_qt_bin}" NO_DEFAULT_PATH)
        if(_qt_deploy)
            add_custom_command(TARGET fpga_sdk_demo POST_BUILD COMMAND "${_qt_deploy}" --no-translations --no-opengl-sw --compiler-runtime "$<TARGET_FILE:fpga_sdk_demo>")
        endif()
    endif()
    if(BUILD_TESTING)
        add_test(NAME sdk_qt_demo_smoke COMMAND fpga_sdk_demo --smoke-test ${CMAKE_CURRENT_BINARY_DIR}/demo_smoke.png)
        set_tests_properties(sdk_qt_demo_smoke PROPERTIES TIMEOUT 45)
    endif()
    install(TARGETS fpga_sdk_demo RUNTIME DESTINATION bin)
    if(WIN32)
        install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/" DESTINATION bin FILES_MATCHING PATTERN "*.dll" PATTERN "CMakeFiles" EXCLUDE)
    endif()
endif()
install(TARGETS fpga_sdk EXPORT FPGASDKTargets RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(EXPORT FPGASDKTargets NAMESPACE FPGA:: DESTINATION lib/cmake/FPGASDK)
install(FILES cmake/FPGASDKConfig.cmake DESTINATION lib/cmake/FPGASDK)
include(CMakePackageConfigHelpers)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/FPGASDKConfigVersion.cmake" VERSION 2.0.0 COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/FPGASDKConfigVersion.cmake" DESTINATION lib/cmake/FPGASDK)
install(TARGETS fpga_hardware_probe RUNTIME DESTINATION bin)
install(TARGETS fpga_hardware_roundtrip RUNTIME DESTINATION bin)
install(DIRECTORY examples/ DESTINATION examples)
install(FILES README.md DESTINATION .)
install(DIRECTORY docs/ DESTINATION docs)
