cmake_minimum_required(VERSION 3.20)
project(FPGASDKConsumer LANGUAGES CXX)
if(MSVC AND DEFINED ENV{FPGA_MSVC_INCLUDE_PREFIX})
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "$ENV{FPGA_MSVC_INCLUDE_PREFIX}")
endif()
find_package(FPGASDK CONFIG REQUIRED)
add_executable(sdk_example main.cpp)
target_link_libraries(sdk_example PRIVATE FPGA::SDK)
target_compile_features(sdk_example PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(sdk_example PRIVATE /utf-8)
endif()
if(WIN32)
    add_custom_command(TARGET sdk_example POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:FPGA::SDK> $<TARGET_FILE_DIR:sdk_example>)
endif()
